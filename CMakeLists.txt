cmake_minimum_required(VERSION 3.20)
project(Wave VERSION 0.2.0 LANGUAGES CXX)

set(WAVE_STANDALONE OFF)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    set(WAVE_STANDALONE ON)
    if(NOT DEFINED CMAKE_MSVC_RUNTIME_LIBRARY)
        set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    include(CTest)
endif()
option(WAVE_BUILD_DEMO "Build the Win32/DX11 demo and TCP adapter" ${WAVE_STANDALONE})
option(WAVE_BUILD_EXAMPLES "Build the independent Windows data sources" ${WAVE_STANDALONE})
set(WAVE_TESTS_DEFAULT OFF)
if(WAVE_STANDALONE AND BUILD_TESTING)
    set(WAVE_TESTS_DEFAULT ON)
endif()
option(WAVE_BUILD_TESTS "Build Wave regression tests" ${WAVE_TESTS_DEFAULT})
if((WAVE_BUILD_DEMO OR WAVE_BUILD_EXAMPLES) AND NOT WIN32)
    message(FATAL_ERROR "Demo/data-source executables require Windows. Disable WAVE_BUILD_DEMO and WAVE_BUILD_EXAMPLES to embed the core.")
endif()

# A parent may provide an existing compatible ImGui/ImPlot target to avoid
# compiling a second copy. Otherwise use the pinned bundled dependencies.
set(WAVE_PLOT_TARGET "" CACHE STRING "Existing target providing compatible ImGui and ImPlot")
if(WAVE_PLOT_TARGET)
    if(NOT TARGET "${WAVE_PLOT_TARGET}")
        message(FATAL_ERROR "WAVE_PLOT_TARGET must name an existing target")
    endif()
    set(WAVE_PLOT_LINK "${WAVE_PLOT_TARGET}")
else()
    set(IMGUI_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/imgui-1.92.9b-docking" CACHE PATH "Dear ImGui source directory")
    set(IMPLOT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/implot-1.0" CACHE PATH "ImPlot source directory")
    if(NOT EXISTS "${IMGUI_DIR}/imgui.cpp" OR NOT EXISTS "${IMPLOT_DIR}/implot.cpp")
        message(FATAL_ERROR "Dependencies are missing. Run scripts/bootstrap.ps1 -SkipToolchain or provide WAVE_PLOT_TARGET.")
    endif()
    add_library(wave_plot STATIC
        "${IMGUI_DIR}/imgui.cpp"
        "${IMGUI_DIR}/imgui_draw.cpp"
        "${IMGUI_DIR}/imgui_tables.cpp"
        "${IMGUI_DIR}/imgui_widgets.cpp"
        "${IMPLOT_DIR}/implot.cpp"
        "${IMPLOT_DIR}/implot_items.cpp")
    target_include_directories(wave_plot PUBLIC "${IMGUI_DIR}" "${IMPLOT_DIR}")
    if(WIN32)
        target_compile_definitions(wave_plot PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
        target_link_libraries(wave_plot PUBLIC user32 imm32 shell32)
    endif()
    set(WAVE_PLOT_LINK wave_plot)
endif()

find_package(Threads REQUIRED)
add_library(wave_data STATIC src/data_core.cpp)
target_include_directories(wave_data PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_link_libraries(wave_data PUBLIC Threads::Threads)
add_library(Wave::data ALIAS wave_data)
add_library(wave_view STATIC src/wave_view.cpp)
target_link_libraries(wave_view PUBLIC wave_data "${WAVE_PLOT_LINK}")
add_library(Wave::view ALIAS wave_view)
set(WAVE_TARGETS wave_data wave_view)
if(TARGET wave_plot)
    list(APPEND WAVE_TARGETS wave_plot)
endif()

if(WAVE_BUILD_DEMO)
    if(WAVE_PLOT_TARGET AND NOT EXISTS "${IMGUI_DIR}/backends/imgui_impl_win32.cpp")
        message(FATAL_ERROR "Set IMGUI_DIR to the matching source tree to build the demo backends.")
    endif()
    add_library(wave_gui STATIC
        "${IMGUI_DIR}/backends/imgui_impl_win32.cpp"
        "${IMGUI_DIR}/backends/imgui_impl_dx11.cpp")
    target_include_directories(wave_gui PUBLIC "${IMGUI_DIR}/backends")
    target_link_libraries(wave_gui PUBLIC "${WAVE_PLOT_LINK}" d3d11 dxgi d3dcompiler dwmapi gdi32)
    add_executable(WaveDemo WIN32 src/main.cpp)
    target_link_libraries(WaveDemo PRIVATE wave_view wave_gui ws2_32)
    list(APPEND WAVE_TARGETS wave_gui WaveDemo)
endif()
if(WAVE_BUILD_EXAMPLES)
    add_executable(SystemSource examples/system_source.cpp)
    add_executable(AudioSource examples/audio_source.cpp)
    target_include_directories(SystemSource PRIVATE src)
    target_include_directories(AudioSource PRIVATE src)
    target_compile_definitions(SystemSource PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_compile_definitions(AudioSource PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(SystemSource PRIVATE ws2_32)
    target_link_libraries(AudioSource PRIVATE ws2_32 ole32 uuid)
    list(APPEND WAVE_TARGETS SystemSource AudioSource)
endif()
foreach(target WaveDemo SystemSource AudioSource)
    if(TARGET ${target})
        set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/build")
        foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
            set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_CURRENT_SOURCE_DIR}/build")
        endforeach()
    endif()
endforeach()

if(WAVE_BUILD_TESTS)
    enable_testing()
    add_executable(data_core_tests tests/data_core_tests.cpp tests/simulation_source.cpp)
    target_link_libraries(data_core_tests PRIVATE wave_data)
    add_test(NAME data_core_tests COMMAND data_core_tests)
    add_executable(display_clock_tests tests/display_clock_tests.cpp)
    target_link_libraries(display_clock_tests PRIVATE wave_data)
    add_test(NAME display_clock_tests COMMAND display_clock_tests)
    add_executable(view_tests tests/view_tests.cpp)
    target_link_libraries(view_tests PRIVATE wave_view)
    add_test(NAME view_tests COMMAND view_tests)
    list(APPEND WAVE_TARGETS data_core_tests display_clock_tests view_tests)
    if(WIN32)
        add_executable(tcp_source_tests tests/tcp_source_tests.cpp)
        target_compile_definitions(tcp_source_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
        target_link_libraries(tcp_source_tests PRIVATE wave_data ws2_32)
        add_test(NAME tcp_source_tests COMMAND tcp_source_tests)
        list(APPEND WAVE_TARGETS tcp_source_tests)
    endif()
endif()

foreach(target IN LISTS WAVE_TARGETS)
    target_compile_features(${target} PUBLIC cxx_std_20)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(${target} PRIVATE /utf-8 /W4 "$<$<CONFIG:Release>:/O2>")
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra "$<$<CONFIG:Release>:-O2>")
    endif()
    if(MINGW)
        get_target_property(target_type ${target} TYPE)
        if(target_type STREQUAL "EXECUTABLE")
            target_link_options(${target} PRIVATE -static)
        endif()
    endif()
endforeach()
